```ts
// frontend/src/lib/courseView.ts
import type { CardOut, ModulePathItem } from "../api/types";

export type PasoId = "leer" | "pronunciar" | "hablar" | "repasar";

export interface PasoVM {
  n: string;
  id: PasoId;
  to: string;
  e: "ahora" | "luego";
}

export interface CourseVM {
  id: "aleman";
  progreso: number;
  vencen: number | null;
  currentModule: ModulePathItem | null;
  pasos: PasoVM[];
}

// ponytail: F1 no tiene máquina de estados de pasos — "ahora" es siempre el
// primero y no existe "hecha"; la máquina real llega con plan_steps en F3.
export function buildCourseVM(args: {
  modules: ModulePathItem[] | null;
  due: CardOut[] | null;
  latestStoryId: string | null;
}): CourseVM {
  const { modules, due, latestStoryId } = args;
  const total = modules?.length ?? 0;
  const done = modules?.filter((m) => m.completed).length ?? 0;
  const leerTo = latestStoryId ? `/story/${latestStoryId}` : "/story/new";
  // ponytail: 02 y 04 apuntan ambos a /review — hoy la cola de Practice mezcla
  // pronunciación y repaso SRS; F2 separa Repaso en su propia pantalla.
  const pasos: PasoVM[] = [
    { n: "01", id: "leer", to: leerTo, e: "ahora" },
    { n: "02", id: "pronunciar", to: "/review", e: "luego" },
    { n: "03", id: "hablar", to: "/chat", e: "luego" },
    { n: "04", id: "repasar", to: "/review", e: "luego" },
  ];
  return {
    id: "aleman",
    progreso: total ? done / total : 0,
    vencen: due ? due.length : null,
    currentModule: modules?.find((m) => m.is_current) ?? null,
    pasos,
  };
}

export type RowEstado = "dominado" | "encurso" | "proximo" | "futuro";

export function moduleRowEstado(
  m: Pick<ModulePathItem, "completed" | "is_current" | "unlocked">
): RowEstado {
  if (m.completed) return "dominado";
  if (m.is_current) return "encurso";
  if (m.unlocked) return "proximo";
  return "futuro";
}
```

```tsx
import { useTranslation } from "react-i18next";
import { api } from "../api/client";
import type { CardOut, ModulePathItem, StoryListItem } from "../api/types";
import { buildCourseVM, type CourseVM } from "../lib/courseView";
import { useGreeting, useMastheadDate } from "../lib/dateLabel";
import { languageLabel } from "../lib/languages";
import { useUser } from "../lib/user";
import KlaraMark from "../components/KlaraMark";

export function useCourseData(): { vm: CourseVM; loading: boolean } {
  const [vm, setVm] = useState<CourseVM>(() =>
    buildCourseVM({ modules: null, due: null, latestStoryId: null })
  );
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    let cancelled = false;
    (async () => {
      const [modules, due, stories] = await Promise.all([
        api.listModules().catch(() => null),
        api.dueCards(50).catch(() => null),
        api.listStories(1, 0).catch(() => [] as StoryListItem[]),
      ]);
      if (cancelled) return;
      setVm(
        buildCourseVM({
          modules: modules as ModulePathItem[] | null,
          due: due as CardOut[] | null,
          latestStoryId: stories[0]?.id ?? null,
        })
      );
      setLoading(false);
    })();
    return () => { cancelled = true; };
  }, []);
  return { vm, loading };
}

export default function Inicio() {
  const { t } = useTranslation();
  const navigate = useNavigate();
  const { user } = useUser();
  const { vm } = useCourseData();
  const fecha = useMastheadDate();
  const hola = useGreeting();
  const paso = vm.pasos.find((p) => p.e === "ahora") ?? vm.pasos[0];
  const titulo = user ? languageLabel(user.target_language) : "…";

  return (
    <main className="m3-page">
      <div className="m3-top">
        <span className="marca"><KlaraMark size={14} /><span className="wordmark">Klara</span></span>
        <span>{fecha}</span>
      </div>
      <h1 className="m4-saludo">{hola}, <em>{user?.display_name ?? "…"}</em>.</h1>
      <div className="m4-sec">{t("biblioteca.titulo")}</div>
      <div className="m4-lib">
        <button className="m4-tema" onClick={() => navigate("/curso/aleman")}>
          <span className="f1">
            <span className="tit">{titulo}</span>
            <span className="estado" data-hot="true">
              {vm.vencen != null && vm.vencen > 0
                ? t("biblioteca.estadoDue", { count: vm.vencen })
                : t("biblioteca.estadoHoy")}
            </span>
          </span>
          <span className="sub">{t("biblioteca.cursoSub", { level: user?.level ?? "" })}</span>
          <span className="m4-bar"><i style={{ width: `${vm.progreso * 100}%` }} /></span>
        </button>
        <button className="m4-add" onClick={() => navigate("/nuevo")}>
          <span className="tile">+</span>
          <span>
            <span className="tit">{t("biblioteca.nuevoTitulo")}</span>
            <span className="sub">{t("biblioteca.nuevoDek")}</span>
          </span>
        </button>
      </div>
      <button className="m3-hero" style={{ marginTop: "auto" }} onClick={() => navigate(paso.to)}>
        <span className="tag">
          <span>{t("biblioteca.heroTag", { curso: titulo })}</span>

          </span>
          <span className="arrow">→</span>
        </span>
      </button>
    </main>
  );
}
```

```tsx
import { useNavigate } from "react-router-dom";
import { useTranslation } from "react-i18next";
import { moduleRowEstado } from "../lib/courseView";
import { useCourseData } from "./Inicio";
import { languageLabel } from "../lib/languages";
import { useUser } from "../lib/user";

export default function MapaCurso() {
  const { t } = useTranslation();
  const navigate = useNavigate();
  const { user } = useUser();
  const { vm, loading } = useCourseData();
  const titulo = user ? languageLabel(user.target_language) : "…";

  const stops = ["A0", "A1", "A2", "B1"]; // ponytail: escala fija del curso de; por-curso llega con F3

  return (
    <main className="m3-page">
      <div className="m4-backrow">
        <button className="m3-back" onClick={() => navigate("/curso/aleman")}>← {titulo}</button>
        <span className="der">{t("mapa.kicker")}</span>
      </div>
      <h1 className="m4-h1">
        {t("mapa.h1pre")} <em>{user?.level ?? ""}</em>.
      </h1>
      <div className="m3-cefr" role="img" aria-label={t("mapa.progressAria", { pct: Math.round(x) })}>
        <span className="aqui" style={{ left: `${x}%` }}>{t("mapa.aqui")}</span>
        <div className="track">
          <div className="fill" style={{ width: `${x}%` }} />
          <span className="dot" style={{ left: `${x}%` }} />
        </div>
        {stops.map((l, i) => {
          const p = (i / (stops.length - 1)) * 100;
          return (
            <span key={l} className="stop" data-past={p <= x ? "true" : "false"} style={{ left: `${p}%` }}>
              {l}
            </span>
          );
        })}
      </div>
      <ul className="m3-rows">
        {vm.modules?.map((m) => (
          <li key={m.id} data-e={moduleRowEstado(m)} onClick={() => navigate(`/module/${m.id}`)} style={{ cursor: "pointer" }}>
            <span className="n">{String(m.sequence_order).padStart(2, "0")}</span>
            <span className="tit">{m.title}</span>
            <span className="estado">
              {m.completed
                ? "✓"
                : m.is_current
                ? t("mapa.encurso", { done: m.stories_finished, total: m.stories_to_complete })
                : m.unlocked
                ? t("mapa.proximo")
                : "—"}
            </span>
          </li>

    </main>
  );
}
```
